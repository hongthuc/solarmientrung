    function filterCategory(catSlug) {
      // Cập nhật trạng thái nút danh mục
      document.querySelectorAll('#homepageCategoryBar .cat-pill').forEach(btn => {
        if (btn.getAttribute('data-cat') === catSlug) {
          btn.className = 'cat-pill active-cat flex items-center justify-center gap-1.5 px-2.5 py-2 rounded-lg border border-solar-blue bg-solar-blue text-white text-[11.5px] sm:text-[12px] font-semibold transition shadow-2xs';
        } else {
          btn.className = 'cat-pill flex items-center justify-center gap-1.5 px-2.5 py-2 rounded-lg border border-slate-200 bg-slate-50 hover:border-solar-blue hover:bg-solar-lightblue text-slate-700 hover:text-solar-blue text-[11.5px] sm:text-[12px] font-semibold transition';
        }
      });

      const sections = document.querySelectorAll('.category-section');
      const items = document.querySelectorAll('.product-item');

      if (catSlug === 'all') {
        sections.forEach(sec => sec.classList.remove('hidden'));
        items.forEach(item => item.classList.remove('hidden'));
        return;
      }

      sections.forEach(sec => {
        const secCats = sec.getAttribute('data-section-cat') || '';
        if (secCats.includes(catSlug)) {
          sec.classList.remove('hidden');
        } else {
          sec.classList.add('hidden');
        }
      });

      items.forEach(item => {
        if (item.getAttribute('data-category') === catSlug) {
          item.classList.remove('hidden');
        } else {
          item.classList.add('hidden');
        }
      });

      document.getElementById('san-pham-chinh').scrollIntoView({ behavior: 'smooth', block: 'start' });
    }

    function filterByKeyword(keyword) {
      const q = keyword.trim().toLowerCase();
      const sections = document.querySelectorAll('.category-section');
      const items = document.querySelectorAll('.product-item');

      sections.forEach(sec => sec.classList.remove('hidden'));
      items.forEach(item => {
        const name = item.getAttribute('data-name') || '';
        if (!q || name.includes(q)) {
          item.classList.remove('hidden');
        } else {
          item.classList.add('hidden');
        }
      });

      document.getElementById('san-pham-chinh').scrollIntoView({ behavior: 'smooth', block: 'start' });
    }

      mobileSlides.forEach((slide, idx) => {
        if (idx === currentSlideIndex) {
          slide.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
          slide.classList.add('opacity-100', 'z-10');
        } else {
          slide.classList.remove('opacity-100', 'z-10');
          slide.classList.add('opacity-0', 'z-0', 'pointer-events-none');
        }
      });

      const dot0 = document.getElementById('sliderDot0');
      const dot1 = document.getElementById('sliderDot1');
      if (currentSlideIndex === 0) {
        dot0.className = 'w-6 h-1.5 rounded-full bg-amber-400 transition-all';
        dot1.className = 'w-2 h-1.5 rounded-full bg-white/50 transition-all';
      } else {
        dot0.className = 'w-2 h-1.5 rounded-full bg-white/50 transition-all';
        dot1.className = 'w-6 h-1.5 rounded-full bg-amber-400 transition-all';
      }
    }

    function changeSlide(step) {
      currentSlideIndex = (currentSlideIndex + step + totalSlides) % totalSlides;
      updateSliderUI();
      resetSlideTimer();
    }

    function goToSlide(index) {
      currentSlideIndex = index;
      updateSliderUI();
      resetSlideTimer();
    }

    function resetSlideTimer() {
      if (slideInterval) clearInterval(slideInterval);
      slideInterval = setInterval(() => {
        currentSlideIndex = (currentSlideIndex + 1) % totalSlides;
        updateSliderUI();
      }, 4500);
    }
